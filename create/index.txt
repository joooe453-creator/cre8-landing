    const token1 = decodeAddress(token1Hex);
    const targetAddress = target.address.toLowerCase();
    const quoteAddress = quote.address.toLowerCase();
    if (!([token0, token1].includes(targetAddress) && [token0, token1].includes(quoteAddress))) throw new Error("Factory pair token mismatch");
    const raw = reservesHex.slice(2);
    const reserve0 = BigInt("0x" + raw.slice(0, 64));
    const reserve1 = BigInt("0x" + raw.slice(64, 128));
    const targetReserve = toTokenNumber(token0 === targetAddress ? reserve0 : reserve1, target.decimals);
    const quoteReserve = toTokenNumber(token0 === quoteAddress ? reserve0 : reserve1, quote.decimals);
    const spotPriceUsd = quoteReserve / targetReserve * quoteUsdPrice;
    const liquidityUsd = quoteReserve * quoteUsdPrice * 2;
    if (!(Number.isFinite(liquidityUsd) && liquidityUsd > 0 && Number.isFinite(spotPriceUsd) && spotPriceUsd > 0)) throw new Error("Pair has no usable liquidity");
    return { protocol: "v2", venue: "PancakeSwap V2", poolAddress, fee: 2500, liquidityUsd, spotPriceUsd };
  }

  async function discoverContractTarget(address, skipKnown, requestedPoolAddress) {
    const normalized = address.toLowerCase();
    const known = S.registry.targets.find((target) => target.address.toLowerCase() === normalized);
    const requestedPool = String(requestedPoolAddress || "").toLowerCase();
    if (requestedPoolAddress && !ADDRESS_RE.test(requestedPool)) throw new Error("Selected pool address is invalid.");
    if (known && !known.dynamic && !skipKnown && !requestedPool) return known;
    if (normalized === S.registry.quoteAsset.address.toLowerCase()) throw new Error("Accounting USDT uses its direct supply destination, without a swap pool. Select USDT from the asset catalog.");
    const chainId = await rpc("eth_chainId");
    if (Number(BigInt(chainId)) !== 56) throw new Error("RPC is not connected to BNB Chain.");
    const blockTag = await rpc("eth_blockNumber");
    const [code, symbolHex, decimalsHex] = await Promise.all([
      rpc("eth_getCode", [normalized, blockTag]),
      ethCall(normalized, "0x95d89b41", blockTag).catch(() => "0x"),
      ethCall(normalized, "0x313ce567", blockTag),
    ]);
    if (!code || code === "0x") throw new Error("No token contract exists at this address.");
    const decimals = Number(BigInt(decimalsHex));
    if (!Number.isInteger(decimals) || decimals < 0 || decimals > 36) throw new Error("Unsupported token decimals.");
    const symbol = decodeText(symbolHex).replace(/[^A-Za-z0-9._-]/g, "").slice(0, 16) || ("TOKEN-" + normalized.slice(-4));
    const target = { address: normalized, decimals };
    const discovery = S.registry.discovery;
    const wrappedNative = S.registry.targets.find((candidate) => candidate.id === "crypto:WBNB");
    if (!wrappedNative?.liquidity?.spotPriceUsd) throw new Error("BNB bridge price is unavailable.");
    const quotes = [
      { symbol: "USDT", address: discovery.quoteAddress, decimals: discovery.quoteDecimals, usdPrice: 1, bridge: null },
      { symbol: "WBNB", address: wrappedNative.address, decimals: wrappedNative.decimals, usdPrice: wrappedNative.liquidity.spotPriceUsd, bridge: wrappedNative.swap },
    ].filter((quote) => quote.address.toLowerCase() !== normalized);
    const checks = [];
    for (const quote of quotes) {
      const v3Addresses = await Promise.all(discovery.v3FeeTiers.map(async (fee) => {
        const data = "0x1698ee82" + addressWord(normalized) + addressWord(quote.address) + uintWord(fee);
        return { fee, poolAddress: decodeAddress(await ethCall(discovery.pancakeV3Factory, data, blockTag)) };
      }));
      const v2Data = "0xe6a43905" + addressWord(normalized) + addressWord(quote.address);
      const v2Address = decodeAddress(await ethCall(discovery.pancakeV2Factory, v2Data, blockTag));
      const decorate = async (promise) => {
        const route = await promise;
        const firstHop = { tokenIn: normalized, tokenOut: quote.address, protocol: route.protocol, poolAddress: route.poolAddress, fee: route.fee };
        route.path = quote.bridge ? [firstHop, ...quote.bridge.path] : [firstHop];
        route.liquidityUsd = quote.bridge ? Math.min(route.liquidityUsd, wrappedNative.liquidity.usd) : route.liquidityUsd;
        route.venue = quote.bridge ? "PancakeSwap via BNB" : route.venue;
        route.protocol = quote.bridge ? "multi-hop" : route.protocol;
        return route;
      };
      v3Addresses.filter((candidate) => candidate.poolAddress !== ZERO_ADDRESS).forEach((candidate) => {
        checks.push(decorate(inspectV3Pool(target, quote, quote.usdPrice, candidate.poolAddress, candidate.fee, blockTag)));
      });
      if (v2Address !== ZERO_ADDRESS) checks.push(decorate(inspectV2Pool(target, quote, quote.usdPrice, v2Address, blockTag)));
    }
    const inspected = await Promise.allSettled(checks);
    const routes = inspected.filter((result) => result.status === "fulfilled").map((result) => result.value);
    if (!routes.length) throw new Error("No initialized Pancake V2/V3 pool to USDT or BNB found.");
    routes.sort((left, right) => right.liquidityUsd - left.liquidityUsd);
    const route = requestedPool
      ? routes.find((candidate) => candidate.path[0]?.poolAddress.toLowerCase() === requestedPool)
      : routes[0];
    if (!route) throw new Error("Selected pool is not an executable USDT/BNB route. Choose another pool or search the CA for automatic routing.");
    return {
      id: known?.id || ("crypto:custom:" + normalized),
      category: known?.category || "crypto",
      symbol,
      displaySymbol: known?.displaySymbol || symbol,
      ticker: known?.ticker || symbol,
      underlyingName: known?.underlyingName,
      address: normalized,
      decimals,
      dynamic: true,
      swap: { venue: route.venue, protocol: route.protocol, quoteSymbol: "USDT", poolAddress: route.poolAddress, fee: route.fee, path: route.path },
      liquidity: {
        usd: Number(route.liquidityUsd.toFixed(2)),
        verifiedAt: new Date().toISOString(),
        verifiedAtBlock: Number(BigInt(blockTag)),
        methodology: "Every route hop was checked against a Pancake factory on BNB Chain; effective TVL uses the route bottleneck and is not guaranteed execution depth.",
      },
      discovery: { source: "BSC JSON-RPC + Pancake factory", exactPair: true, customToken: true, verifiedAtBlock: Number(BigInt(blockTag)) },
      routeSelection: requestedPool
        ? { mode: "pinned-pool", poolAddress: route.path[0].poolAddress, protocol: route.path[0].protocol, fee: route.path[0].fee }
        : { mode: "best-liquidity" },
    };
  }

  function equalize() {
    const ids = [...S.selected.keys()];

      row.weightBps = base + (remainder-- > 0 ? 1 : 0);

  function problems() {
    const out = [];
    if (!S.registry) out.push(["s-1", "Verified market registry is unavailable."]);
    if (!S.selected.size) out.push(["s-1", "Select at least one target."]);
    if (S.registry && [...S.selected.keys()].some((id) => !S.registry.targets.some((target) => target.id === id))) out.push(["s-1", "A selected target is no longer available."]);
    if (S.registry && [...S.selected.keys()].some((id) => {
      const target = S.registry.targets.find((candidate) => candidate.id === id);
      return target?.dynamic && !(target.discovery?.exactPair && target.swap?.path?.length && target.liquidity?.usd > 0);
    })) out.push(["s-1", "A custom token route has not passed the on-chain pool checks."]);
    // The live read mirrors what the vault refuses on chain; a refused route cannot be part of the design.
    for (const [id, row] of S.selected.entries()) {
      const target = S.registry?.targets.find((candidate) => candidate.id === id);
      if (!target?.top30) continue;
      if (!target.top30.feed) out.push(["s-1", `${displayName(target)} has no reviewed Chainlink USD feed, so the vault cannot price it.`]);
      else if (liveStatus(target, row.marketId)?.ok === false) out.push(["s-1", `${displayName(target)}${row.marketId ? " lending route" : ""}: ${liveStatus(target, row.marketId).reason}.`]);
    }
    const total = [...S.selected.values()].reduce((sum, row) => sum + row.weightBps, 0);
    const caps = [...S.selected.values()].map((row) => row.capBps);
    if (caps.some((cap) => !Number.isInteger(cap) || cap < 0 || cap > 9500)) out.push(["s-1", "Each investment limit must be 0–95% of vault value, with at most two decimal places."]);
    if (caps.reduce((sum, cap) => sum + cap, 0) > 9500) out.push(["s-1", "Investment limits must total at most 95%; keep at least 5% idle cash."]);
    if (total !== 10000) out.push(["s-1", `Strategy weights must total 100% (currently ${(total / 100).toFixed(2)}%).`]);
    if ([...S.selected.values()].some((row) => !(row.weightBps > 0))) out.push(["s-1", "Every selected target needs a positive weight."]);
    for (const [id, row] of S.selected.entries()) {
      const target = S.registry?.targets.find((candidate) => candidate.id === id);
      if (target?.accountingAsset && !row.marketId) out.push(["s-2", "Accounting USDT must use its exact supply destination; uninvested USDT stays in the idle reserve."]);
      if (!row.marketId) continue;
      const market = compatibleMarkets(id).find((candidate) => candidate.id === row.marketId);
      if (!market) {
        out.push(["s-2", "A selected destination is not a verified exact-asset market."]);
      }
    }
    if (S.registry && S.mode !== "agent-managed" && num($("#seed").value) > 0) {
      const deploymentBps = S.mode === "lump-sum" ? 9500 : Math.round(num($("#tranche").value) * 100);
      for (const [id, row] of S.selected.entries()) {
        const target = S.registry.targets.find((candidate) => candidate.id === id);
        const initialOrderUsd = num($("#seed").value) * deploymentBps / 10000 * row.weightBps / 10000;
        const status = target?.top30 ? liveStatus(target, row.marketId) : null;
        if (status) {
          if (status.ok && initialOrderUsd > status.maxOrderUsd) out.push(["s-1", `${displayName(target)} initial order ${formatUsd(initialOrderUsd)} is above the live max ${formatUsd(status.maxOrderUsd)} per order (0.1% of pool depth).`]);
          continue;
        }
        // Before the first live read (or if it failed), the snapshot TVL bounds the order.
        const capacityUsd = (target && legLiquidity(target, row.marketId))?.usd * 0.001;
        if (Number.isFinite(capacityUsd) && initialOrderUsd > capacityUsd) {
          out.push(["s-1", `${displayName(target)} initial order exceeds the 0.1% pool-TVL limit.`]);
        }
      }
    }
    if (S.mode === "accumulate") {
      if (!$("#schedule-on").checked && !$("#dip-on").checked) out.push(["s-2", "Enable a schedule, a drawdown trigger, or both."]);
      if ($("#schedule-on").checked && !validCadence($("#interval").value, $("#interval-unit").value)) out.push(["s-2", "Choose a valid schedule: up to 365 days, 52 weeks or 12 months per interval."]);
      const startAt = Date.parse($("#schedule-start").value);
      if ($("#schedule-on").checked && (!Number.isFinite(startAt) || startAt <= Date.now())) out.push(["s-2", "Schedule start must be in the future."]);
      if ($("#schedule-on").checked && $("#schedule-end").value === "after-executions" && !(num($("#schedule-count").value) >= 1 && num($("#schedule-count").value) <= 1000)) out.push(["s-2", "Scheduled buy count must be between 1 and 1,000."]);
      if ($("#schedule-on").checked && $("#schedule-end").value === "on-date") {
        const endAt = Date.parse($("#schedule-end-date").value);
        if (!Number.isFinite(endAt) || endAt <= startAt) out.push(["s-2", "Schedule end must be after the start."]);
      }
      if ($("#dip-on").checked && !(num($("#dip").value) >= 3 && num($("#dip").value) <= 30)) out.push(["s-2", "Dip trigger must be between 3% and 30%."]);
      if (tranchePercentageNeeded() && !(num($("#tranche").value) >= 1 && num($("#tranche").value) <= 25)) out.push(["s-2", "Tranche size must be between 1% and 25%."]);
    }
    if ($("#rebalance-on").checked) {
      if (!(num($("#rebalance-drift").value) >= 5 && num($("#rebalance-drift").value) <= 20)) out.push(["s-2", "Rebalance drift must be between 5% and 20%."]);
    }
    if (!$("#name").value.trim()) out.push(["s-3", "Name the vault."]);
    if (!(num($("#pf").value) >= 0 && num($("#pf").value) <= 20)) out.push(["s-3", "Performance fee must be 0–20%."]);
    if (num($("#mf").value) !== 0) out.push(["s-3", "Management fees are not supported by this vault."]);
    if (num($("#seed").value) !== 100) out.push(["s-3", "The creator seed is exactly 100 USDT."]);
    return out;
  }

  function finiteSchedule() {
    return S.mode === "accumulate" && $("#schedule-on").checked && $("#schedule-end").value !== "until-stopped";
  }
  function tranchePercentageNeeded() {
    return S.mode === "accumulate" && (!finiteSchedule() || $("#dip-on").checked);
  }

    const finite = finiteSchedule();
    $("#tranche-field").classList.toggle("hide", !tranchePercentageNeeded());

    $("#tranche-label").textContent = finite ? "Dip tranche · 1–25% of available cash" : "Tranche size · 1–25% of available cash";
  }
  function refresh() {
    updateTrancheControls();
    const total = [...S.selected.values()].reduce((sum, row) => sum + row.weightBps, 0);
    const marketCount = [...S.selected.values()].filter((row) => row.marketId).length;
    const errors = problems();

    $("#symbol").textContent = name ? `av${name.split(/\s+/).map((word) => word.replace(/[^a-z0-9]/gi, "")[0] || "").join("").toUpperCase().slice(0, 5)}` : "—";
    $("#weight-total").textContent = `${(total / 100).toFixed(total % 100 ? 2 : 0)}%`;
    const capTotal = [...S.selected.values()].reduce((sum, row) => sum + row.capBps, 0);
    const capLabel = Number.isFinite(capTotal) ? `${(capTotal / 100).toFixed(capTotal % 100 ? 2 : 0)}%` : "—";
    const cashLabel = Number.isFinite(capTotal) ? `${Math.max(0, 100 - capTotal / 100).toFixed(capTotal % 100 ? 2 : 0)}%` : "—";
    $("#selected-count").textContent = String(S.selected.size);

    $("#summary-limit-total").textContent = `${capLabel} / 95%`;
    $("#summary-limit-total").classList.toggle("planning-note", capTotal > 9500);
    $("#summary-limit-bar").style.width = `${Number.isFinite(capTotal) ? Math.min(100, Math.max(0, capTotal / 95)) : 0}%`;
    $("#summary-cash").textContent = cashLabel;
    $("#summary-fee").textContent = `${$("#pf").value || "—"}% total`;
    [1, 2, 3].forEach((step) => {
      const stepErrors = errors.filter(([section]) => section === `s-${step}`);
      $(["#step-one-issues", "#step-two-issues", "#step-three-issues"][step - 1]).innerHTML = stepErrors.map(([, message]) => `<li class="bad"><i></i><span>${esc(message)}</span></li>`).join("");
      $(`[data-create-next="${step + 1}"]`).disabled = errors.some(([section]) => Number(section.slice(2)) <= step);
    });
    $$('[data-cap]').forEach((input) => input.setAttribute("aria-invalid", String(!Number.isInteger(S.selected.get(input.dataset.cap).capBps) || capTotal > 9500)));
    $$('[data-weight]').forEach((input) => input.setAttribute("aria-invalid", String(total !== 10000 || !(S.selected.get(input.dataset.weight).weightBps > 0))));
    $("#create-summary-positions").innerHTML = (S.registry ? [...S.selected.entries()].map(([id, row]) => {
      const target = S.registry.targets.find((candidate) => candidate.id === id);
      const market = compatibleMarkets(id).find((candidate) => candidate.id === row.marketId);
      return `<div class="create-summary-position"><b>${esc(target.symbol)}</b><span>${market ? venueLogo(market.protocol) : ""}${esc(market ? `${market.protocol} · ${market.kind === "supply-market" ?   function parseTokenSearchPages(pages, rawQuery) {
    const query = normalizeTokenQuery(rawQuery);
    const rows = [];
    pages.forEach((page) => {
      const included = new Map((Array.isArray(page?.included) ? page.included : []).map((item) => [item.id, item]));
      (Array.isArray(page?.data) ? page.data : []).forEach((pool) => {
        const dexId = pool?.relationships?.dex?.data?.id || "";
        if (!["pancakeswap-v2-bsc", "pancakeswap-v3-bsc"].includes(dexId)) return;
        const attributes = pool?.attributes || {};
        const poolName = String(attributes.name || "");
        const feeMatch = poolName.match(/(\d+(?:\.\d+)?)%\s*$/);
        const version = dexId === "pancakeswap-v2-bsc" ? "V2" : "V3";
        const feePct = version === "V2" ? 0.25 : positiveNumber(feeMatch?.[1]);
        const baseTokenId = pool?.relationships?.base_token?.data?.id;
        [baseTokenId, pool?.relationships?.quote_token?.data?.id].forEach((tokenId) => {
          const token = included.get(tokenId);
          const address = String(token?.attributes?.address || "").toLowerCase();
          const name = String(token?.attributes?.name || "").trim().slice(0, 80);
          const symbol = String(token?.attributes?.symbol || "").trim().slice(0, 24);
          if (!ADDRESS_RE.test(address) || ![name, symbol].some((value) => normalizeTokenQuery(value).includes(query))) return;
          const isBaseToken = tokenId === baseTokenId;
          rows.push({
            address,
            name: name || symbol,
            symbol: symbol || "TOKEN",
            poolAddress: String(attributes.address || "").toLowerCase(),
            version,
            feePct,
            marketCapUsd: isBaseToken ? positiveNumber(attributes.market_cap_usd) : null,
            fdvUsd: isBaseToken ? positiveNumber(attributes.fdv_usd) : null,
            liquidityUsd: positiveNumber(attributes.reserve_in_usd) || 0,
          });
        });
      });
    });
    const deepestByAddress = new Map();
    rows.forEach((row) => {
      const current = deepestByAddress.get(row.address);
      const hasValuation = row.marketCapUsd !== null || row.fdvUsd !== null;
      const currentHasValuation = current && (current.marketCapUsd !== null || current.fdvUsd !== null);
      if (!current || (hasValuation && !currentHasValuation) || (hasValuation === currentHasValuation && row.liquidityUsd > current.liquidityUsd)) {
        deepestByAddress.set(row.address, row);
      }
    });
    rows.forEach((row) => {
      const reference = deepestByAddress.get(row.address);
      row.marketCapUsd = reference.marketCapUsd;
      row.fdvUsd = reference.fdvUsd;
    });
    return rows.sort((left, right) => {
      const rightCap = right.marketCapUsd ?? right.fdvUsd ?? -1;
      const leftCap = left.marketCapUsd ?? left.fdvUsd ?? -1;
      return rightCap - leftCap || right.liquidityUsd - left.liquidityUsd || left.symbol.localeCompare(right.symbol);
    });
  }
