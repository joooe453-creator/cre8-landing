      if (client.activeWalletProvider() !== inventoryProvider || account !== inventoryAccount) throw new Error("Wallet context changed during inventory read");
      const [chain, accounts, header] = await Promise.all([
        inventoryProvider.request({ method: "eth_chainId", params: [] }),
        inventoryProvider.request({ method: "eth_accounts", params: [] }),
        inventoryProvider.request({ method: "eth_getBlockByNumber", params: [blockTag, false] }),

      if (Number(BigInt(chain)) !== deployment.chainId || accounts[0]?.toLowerCase() !== inventoryAccount?.toLowerCase() || header?.hash?.toLowerCase() !== snapshot.blockHash?.toLowerCase()) throw new Error("Inventory chain, account or canonical block changed");

        inventoryProvider.request({ method: "eth_call", params: [{ to: token, data: "0x70a08231" + holder.slice(2).padStart(64, "0") }, blockTag] }),
        inventoryProvider.request({ method: "eth_call", params: [{ to: token, data: "0x313ce567" }, blockTag] }),
      ]);
      const places = Number(BigInt(precision));
      if (!Number.isInteger(places) || places < 0 || places > 36) throw new Error("Unsupported token precision");
      return decimal(BigInt(raw), places);
    };
    const entries = [{ custody: "Fund idle cash", token: snapshot.asset, holder: vaultAddress() }, { custody: "Portfolio cash", token: snapshot.asset, holder: snapshot.portfolio }];
    snapshot.template.legs.forEach((leg) => {

    });
    try {
      await verifyInventoryContext();
      const amounts = await Promise.all(entries.map((entry) => balance(entry.token, entry.holder)));
      await verifyInventoryContext();
      if (snapshot !== live || !panel.isConnected) return;
      $("#positions tbody").innerHTML = entries.map((entry, index) => `<tr><td><strong>${esc(entry.custody)}</strong><span class="position-venue">${addressView(entry.token)}</span></td><td class="position-form">ERC-20 balanceOf<span class="position-mobile-balance">Block ${snapshot.blockNumber}</span></td><td class="r position-balance">${snapshot.blockNumber}</td><td class="r position-value"><strong title="${esc(amounts[index])}">${esc(displayNumber(amounts[index]))}</strong><span class="position-mobile-unit">token units</span></td></tr>`).join("");
